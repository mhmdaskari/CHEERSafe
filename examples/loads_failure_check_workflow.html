    import numpy as np
from cheersafe.inventory import BuildingInventory

def demo_building():
    return BuildingInventory(buildings_file_name="BuildingGeometry_Old.json").buildings[0]

def demo_resistances(building):
    num_panel = int(np.sum(building.NumPanel))
    num_connection = int(np.sum(building.NumConnection))
    num_opening = int(
        building.DoorFront
        + building.DoorBack
        + building.DoorGarage
        + building.OpeningFront
        + building.OpeningBack
        + 2 * (building.OpeningSideH + building.OpeningSideR)
    )
    return {
        "rRSt": np.full(num_panel, 225.0),
        "rRCt": np.full(num_panel, 70.0),
        "rRWCt": np.full(num_connection, 2825.0),
        "rOt": np.full(num_opening, 150.0),
        "rWt": np.full(12, 3000.0),
    }

manual.PressureZoneCoordinate, manual.NumZone = manual.fn_pressure_zone_coordinate()
manual.PressureZoneArea, manual.PanelArea = manual.fn_preprocess_rs()
manual.qh = 0.00256 * 0.85 * manual.WindSpeed ** 2
manual.pRS = manual.fn_initial_load_rs()
manual.pRC = manual.fn_initial_load_rc()
manual.pRWC = manual.fn_initial_load_rwc()
manual.pO = manual.fn_initial_load_o()
manual.pW = manual.fn_initial_load_w()
initial_resistance = manual.fn_initial_failure_check()
manual.rRSt = initial_resistance["Roof Sheathings"]
manual.rRCt = initial_resistance["Roof Coverings"]
manual.rRWCt = initial_resistance["Roof-Wall Connections"]
manual.rOt = initial_resistance["Openings"]
manual.rWt = initial_resistance["Walls"]
manual.pInternal = manual.fn_pressure_internal()
manual.pfRC = manual.fn_final_load_rc()
manual.pfRS = manual.fn_final_load_rs() if manual.pInternal else manual.pRS
manual.pfRWC = manual.fn_final_load_rwc() if manual.pInternal else manual.pRWC
manual.pfO = manual.fn_final_load_o() if manual.pInternal else manual.pO
manual.pfW = manual.fn_final_load_w() if manual.pInternal else manual.pW
manual.rRWCt, manual.RWCFails = manual.fn_final_failure_check_rwc()

print(f"internal pressure: {manual.pInternal:.3f}")
print(f"RWC failure flags: {manual.RWCFails}")
