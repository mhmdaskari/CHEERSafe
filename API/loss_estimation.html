    Loss Estimation#

    The loss estimation module provides functionality for calculating various types of losses including building, component, and regional losses.

    Class Diagram#

    classDiagram
    class BaseLossEstimator {
        +Dict hurricane_data
        +np.ndarray damage_states
        +np.ndarray wind_speeds
        +np.ndarray flood_depths
        +get_adjusted_hazard_data()
    }
    class DamageProbabilityProvider {
        +np.ndarray damage_states
        +Dict prob_damage_states
        +load_damage_state_probabilities()
        +calculate_damage_ratio()
    }
    class BuildingLossEstimator {
        +pd.DataFrame building_values
        +DamageProbabilityProvider damage_provider
        +np.ndarray building_losses
        +calculate_building_losses()
    }
    class ComponentLossEstimator {
        +BuildingLossEstimator building_loss_estimator
        +np.ndarray component_losses
        +calculate_component_losses()
        +calculate_loss_ratios()
    }
    class RegionalLossEstimator {
        +ComponentLossEstimator component_loss_estimator
        +np.ndarray inventory
        +Dict hurricane_losses
        +calculate_regional_losses()
        +combine_wind_flood_losses()
    }
    class LossEstimation {
        +DamageProbabilityProvider damage_provider
        +BuildingLossEstimator building_estimator_wind
        +BuildingLossEstimator building_estimator_flood
        +ComponentLossEstimator component_estimator_wind
        +ComponentLossEstimator component_estimator_flood
        +RegionalLossEstimator regional_estimator
        +estimate_building_losses()
        +estimate_component_losses()
        +estimate_regional_losses()
        +estimate_all_losses()
    }
