def create_cov_matrix(num_component, temp_mean, temp_cv, inter_correlation=0.2, intra_correlation=0.4) -> np.ndarray:
    """Computes the covariance matrix for a set of components.

    This method calculates a covariance matrix based on the number of components,
    their mean values, and coefficient of variation (CV). It considers both
    inter-component and intra-component correlations.

    Args:
        num_component (list[int]): Number of elements in each component.
        temp_mean (numpy.ndarray): Mean values for each element.
        temp_cv (numpy.ndarray): Coefficient of variation for each element.
        inter_correlation (float, optional): Correlation coefficient between elements of different components. Defaults to 0.2.
        intra_correlation (float, optional): Correlation coefficient between elements of the same component. Defaults to 0.4.

    Returns:
        (np.ndarray): Covariance matrix of the components.
    """

    # Calculate total number of components
    total_num = sum(num_component)

    # Create Correlation Matrix
    my_corr = np.full((total_num, total_num), inter_correlation)
    start_idx = 0
    for comp_size in num_component:
        end_idx = start_idx + comp_size
        my_corr[start_idx:end_idx, start_idx:end_idx] = intra_correlation
        start_idx = end_idx
    np.fill_diagonal(my_corr, 1)

    # Calculate Std
    my_std = temp_mean * temp_cv

    # Calculate COV
    std_matrix = np.outer(my_std, my_std)
    sigma1 = my_corr * std_matrix
    np.fill_diagonal(sigma1, my_std ** 2)

    return sigma1

def generate_lhs_samples_general(n, distributions_details, mean, covariance, seed=None) -> np.ndarray:
    """Generates Latin Hypercube samples for specified distributions.

    Args:
        n (int): The number of samples to generate.
        distributions_details (list[dict]): A list of dictionaries specifying the distribution types and parameters for each variable. Each dictionary should have a 'type' key indicating the distribution type (e.g., 'norm', 'lognorm'), and may include additional keys for distribution parameters.
        mean (list[float]): Mean values for the distributions.
        covariance (list[list[float]]): Covariance matrix for the distributions.
        seed (int, optional): Seed for reproducibility. Defaults to None.

    Returns:
        (numpy.ndarray): An array of generated Latin Hypercube samples.

    Example:
        # Example usage for specifying different distributions and parameters
        Option 1:
            distributions_details = [
                {'type': 'norm'},
                {'type': 'lognorm'},
                {'type': 'lognorm', 's': 1.2}
            ]
            mean = [3, 4, 2]
            covariance = [[1, 0.5, 0.3], [0.5, 1, 0.2], [0.3, 0.2, 1]]

        Option 2:
            distributions_details = [{'type': 'norm'}]  # All variables use the 'norm' distribution
            mean = [3, 4, 2]
            covariance = [[1, 0.5, 0.3], [0.5, 1, 0.2], [0.3, 0.2, 1]]

            samples = cls.generate_lhs_samples_general(100, distributions_details, mean, covariance, seed=42, workers=4)
            print(samples)
    """

    dim = len(mean)

    # Check for consistent length
    if len(distributions_details) != dim:
        if len(distributions_details) == 1:
            # If only one distribution detail is provided, replicate it for all variables
            distributions_details = distributions_details * dim
        else:
            raise ValueError(
                "The number of distribution details must either be 1 or match the number of mean values.")

    # Construct distributions
    distributions = []
    for i, dist_details in enumerate(distributions_details):
        dist_type = dist_details.pop('type', None)  # Extract type and remove from the dictionary
        if not dist_type or not hasattr(st, dist_type):
            raise ValueError(f"The distribution type '{dist_type}' is not recognized or missing.")

        params = {'loc': mean[i], 'scale': np.sqrt(covariance[i][i])}  # extract mean and stdev
        params.update(dist_details)  # add any additional parameters provided by user for this distribution
        distributions.append(getattr(st, dist_type)(**params))

    # Generate Latin Hypercube samples using scipy's LatinHypercube
    lhs = LatinHypercube(d=dim, seed=seed)
    lhs_samples = lhs.random(n=n, workers=Config.NUM_OF_WORKERS_SAMPLING)

    # Convert to quantiles of the specified distributions
    for j in range(dim):
        lhs_samples[:, j] = distributions[j].ppf(lhs_samples[:, j])

    # If covariance is provided and is not an identity matrix, transform samples
    if covariance is not None and not np.array_equal(covariance, np.eye(dim)):
        # Convert samples to standard normal quantiles
        norm_samples = np.array([st.norm.ppf(distributions[j].cdf(lhs_samples[:, j])) for j in range(dim)]).T

        # Use Cholesky decomposition for desired covariance
        cholesky = np.linalg.cholesky(covariance)
        transformed_samples = np.dot(norm_samples, cholesky.T)

        # Convert back from standard normal quantiles to the desired distributions
        for j in range(dim):
            lhs_samples[:, j] = distributions[j].ppf(st.norm.cdf(transformed_samples[:, j]))

    return lhs_samples

def generate_lhs_samples_norm(n, mean, covariance, seed=None) -> np.ndarray:
    """Generates n Latin Hypercube Samples with a given mean and covariance matrix.

    Args:
        n (int): The number of samples to generate.
        mean (array-like): An array of means for each variable.
        covariance (array-like): The covariance matrix of the variables.
        seed (optional, int): A seed for the random number generator. Defaults to None.

    Returns:
        np.ndarray: An array of shape (n, len(mean)) with the generated samples.
    """

    # Initialize the LHS sampler
    dim = len(mean)
    lhs = qmc.LatinHypercube(d=dim, seed=seed)
    lhs_samples = lhs.random(n=n, workers=Config.NUM_OF_WORKERS_SAMPLING)

    # Transform the uniform LHS samples to standard normal samples
    norm_samples = norm.ppf(lhs_samples)

    # Apply the Cholesky decomposition to get the transformation matrix
    L = cholesky(covariance, lower=True)

    # Transform the standard normal LHS samples to have the desired mean and covariance
    lhs_samples = mean + np.dot(norm_samples, L.T)

    return lhs_samples

def filter_column_values_df(cls, df: pd.DataFrame, column_name: str, **kwargs) -> np.ndarray:
    """Filters a pandas DataFrame based on the conditions provided in the keyword arguments,
    selects a specified column, and returns it as a numpy array.

    This method allows for dynamic filtering of a DataFrame based on multiple conditions. Each condition is applied to a specified column, and the results are used to filter the DataFrame. After filtering, values from a specified column are extracted and returned as a numpy array.

    Args:
        df (pd.DataFrame): The DataFrame to be filtered.
        column_name (str): The name of the column to be selected after filtering.
        **kwargs (dict): Variable length keyword arguments, where each keyword represents a column in the DataFrame and the corresponding value represents the condition for filtering. Conditions should be provided in a way that they can be evaluated directly in a query.

    Returns:
        np.ndarray: The selected column values after filtering, converted to a numpy array.
    """
    query_list = []
    for key, value in kwargs.items():
        if value is not None:
            query_list.append(f"{key}==@kwargs['{str(key)}']")
    query_all = ' & '.join(query_list)

    return np.array(df.query(query_all)[column_name].values[0])

def from_json(cls, path_to_file: str, dict_key: str = None) -> Union[np.ndarray, Dict[str, Any]]:
    """Reads data from a JSON file and returns it as a numpy array or dictionary.

    This method opens and reads a JSON file specified by `path_to_file`. If the JSON contains a list, it returns a numpy array of that list. If the JSON contains a dictionary and `dict_key` is specified, it returns a numpy array of the data associated with `dict_key`. If `dict_key` is not specified, it returns the entire dictionary.

    Args:
        path_to_file (str): Path to the location of the JSON file.
        dict_key (str, optional): The specific key to extract data from within the JSON file. If not provided and the JSON is a dictionary, the entire dictionary is returned.

    Returns:
        Union[np.ndarray, dict]: Depending on the structure of the JSON and the presence of `dict_key`, returns either a numpy array or a dictionary containing the JSON file data.

    """
    with open(path_to_file) as f:
        json_file = json.load(f)
        if isinstance(json_file, list):
            data = np.array(json_file)
        elif isinstance(json_file, dict):
            if dict_key is not None:
                data = np.array(json_file[dict_key])
            else:
                data = json_file

    return data
