    def calculate_fragility_curves(self):
    """
    Calculate fragility curves for different damage states.

    Fragility curves represent the probability of exceeding a certain damage state
    given a specific wind speed.
    """
    # Get cumulative sum of damage state probabilities (assuming flood depth index 0)
    prob_ds_wind = self.prob_damage_states[(self.building_type, self.resistance_type)][:, 0, :]
    prob_ds_cumsum = np.cumsum(prob_ds_wind, axis=1)

    # Find indices for each damage state group based on damage ratios
    ds_indices = []

    for threshold in self.ds_thresholds:
        indices = np.where(self.damage_state[:, 1] >= threshold)[0]
        if len(indices) > 0:
            ds_indices.append(indices[0])

    # Initialize fragility curves dictionary with wind speed
    self.fragility_curves = {'wind_speed': self.wind_speed[:, 1]}  # Wind speed in mph

    # Calculate DS0 fragility (special case - before first threshold)
    fragility_ds0 = np.mean(prob_ds_cumsum[:, :ds_indices[0]], axis=1)
    fragility_ds0 = 1 - fragility_ds0
    self.fragility_curves['DS0'] = fragility_ds0

    # Calculate fragility for each remaining damage state
    for i in range(len(ds_indices)):
        ds_name = f'DS{i+1}'

        if i < len(ds_indices) - 1:
            # For intermediate damage states
            fragility = np.mean(prob_ds_cumsum[:, ds_indices[i]:ds_indices[i+1]], axis=1)
        else:
            # For the last damage state
            fragility = np.mean(prob_ds_cumsum[:, ds_indices[i]:], axis=1)

        fragility = 1 - fragility

        # Ensure non-negative values for the last damage state
        if i == len(ds_indices) - 1:
            fragility = np.maximum(np.zeros_like(fragility), fragility)

        self.fragility_curves[ds_name] = fragility

    def plot_fragility_curves(self, save_path: Optional[str] = None) -> plt.Figure:
    """
    Plot the fragility curves

    Parameters:
        save_path (str, optional): Path to save the plot
    """
    plt.figure(figsize=(10, 6))

    # Plot DS0 (special case)
    plt.plot(self.fragility_curves['wind_speed'], self.fragility_curves['DS0'], 
             self._get_line_style(0), label='DS0, >=0')

    # Plot remaining damage states
    for i in range(len(self.ds_thresholds)):
        ds_name = f'DS{i+1}'
        threshold = self.ds_thresholds[i]

        # Get appropriate line style (using helper method)
        style = self._get_line_style(i+1)

        plt.plot(self.fragility_curves['wind_speed'], self.fragility_curves[ds_name],
                 style, label=f'DS{i+1}, >={threshold*100:.0f}%')

    # Plot wind scale lines using class variable
    for x in self.wind_scale_lines:
        plt.axvline(x=x, linestyle='--', color='gray')

    plt.title(f'Building Fragility, m={self.building_type}, c={self.resistance_type}')
    plt.xlabel('Wind Speed (mph) 3-second peak gusts')
    plt.ylabel('Probability [DS>=ds|v]')
    plt.axis([min(self.wind_speed[:, 1]), max(self.wind_speed[:, 1]), 0, 1])
    plt.legend(loc='best')

    if save_path:
        plt.savefig(save_path)

    return plt.gcf()

    def plot_analysis_results(self, save_path: Optional[str] = None) -> List[plt.Figure]:
    """
    Plot both vulnerability and fragility curves

    Parameters:
        save_path (str, optional): Folder to save the plots

    Returns:
        List[plt.Figure]: List of figures created
    """
    fig_vulnerability = self.plot_vulnerability_curve()
    fig_fragility = self.plot_fragility_curves()

    if save_path:
        os.makedirs(save_path, exist_ok=True)
        vuln_path = os.path.join(
            save_path, f"vulnerability_m{self.building_type}_c{self.resistance_type}.png"
        )
        frag_path = os.path.join(
            save_path, f"fragility_m{self.building_type}_c{self.resistance_type}.png"
        )
        fig_vulnerability.savefig(vuln_path)
        fig_fragility.savefig(frag_path)

    # Show each plot separately
    plt.figure(fig_vulnerability.number)
    plt.show()
    plt.figure(fig_fragility.number)
    plt.show()

    return [fig_vulnerability, fig_fragility]

    def save_analysis_results(self, file_name: Optional[str] = None) -> str:
    """
    Save analysis results to file

    Parameters:
        file_name (str, optional): File name to save results, defaults to auto-generated name

    Returns:
        str: Path where results were saved
    """
    if file_name is None:
        file_name = f'VulnerabilityAnalysis_m{self.building_type}_c{self.resistance_type}.json'

    file_path = f'{self.output_dir}/{file_name}'

    analysis_results = self.generate_analysis()

    # Convert numpy arrays to lists for JSON serialization
    for key, value in analysis_results['vulnerability_curve'].items():
        if isinstance(value, np.ndarray):
            analysis_results['vulnerability_curve'][key] = value.tolist()

    for key, value in analysis_results['fragility_curves'].items():
        if isinstance(value, np.ndarray):
            analysis_results['fragility_curves'][key] = value.tolist()

    # Save to file using Utils.to_json
    from ..tools import Utils
    Utils.to_json(analysis_results, file_path)

    return file_path

    def calculate_vulnerability_curve(self):
    """
    Calculate vulnerability curve for the specified building type and resistance type.

    The vulnerability curve shows the expected damage ratio for different wind speeds.
    """
    # Load probability of damage states for this building and resistance type
    # In the MATLAB code, it loads from a file, but we already have it loaded
    # We need to extract the specific data for this building and resistance type

    # Filter to get probabilities for wind damage (assuming flood depth index 0)
    prob_ds_wind = self.prob_damage_states[(self.building_type, self.resistance_type)][:, 0, :]

    # Calculate expected damage ratio by multiplying damage state probabilities with damage ratios
    self.expected_damage_ratio = np.dot(prob_ds_wind, self.damage_state[:, 1])

    # Store vulnerability curve data
    self.vulnerability_curve = {
        'wind_speed': self.wind_speed[:, 1],  # Wind speed in mph
        'expected_damage_ratio': self.expected_damage_ratio
    }
