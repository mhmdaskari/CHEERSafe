    def load_replacement_ratio(self, path: str = 'Inputs/ReplacementRatio.json'):
    """
    Load the replacement ratio for each component.

    This method updates the `ReplacementRatio` attribute of the instance by loading data from a JSON file. The components are divided into two categories: Explicit and Implicit:
        * Explicit components include: [RC, RS, RWC, O, W]
        * Implicit components include: [Interior, Electrical, Mechanical, Plumbing]

    Parameters:
        self (BaseDamageRatio): An instance of the BaseDamageRatio class.
        path (str): The path to the JSON file.

    Info: This method does not return any value and updates the following class attribute:
        * `ReplacementRatio`

    Note:
        The `ReplacementRatio` attribute, is a dictionary mapping integers to dictionaries. These dictionaries map strings to floats, representing the replacement ratio for each component.
    """
    matrix = Utils.from_json(path)

    # Column names
    columns = ["Roof Coverings", "Roof Sheathings", "Roof-Wall Connections", "Openings",
               "Walls", "Interior", "Electrical", "Mechanical", "Plumbing"]

    self.ReplacementRatio = {i: dict(zip(columns, row)) for i, row in enumerate(matrix, start=1)}

    def fn_damage_ratio_wind(self):
    """
    This method calculates the damage ratio for each component. The damage ratio is calculated as the number of resistance values that are equal to 0 divided by the total number of resistance values. For RC and RS components, there is a threshold of 35%. If the damage ratio is greater than 35%, then the component is considered to be totaled. For walls, a new damage ratio calculation is used. The wall is considered to be failed if any of the four walls fail. If two walls fail, then the wall is considered to be totaled.

    Attributes:
    -----------
    self:
        ResistanceValues: dict[str, np.ndarray]
            A dictionary that holds the resistance values for each component. The keys are the component types and the values are numpy arrays of resistance values.
        ExplicitDamageRatio: dict[str, float]
            A dictionary that holds the explicit damage ratio for each component. The keys are the component types and the values are the damage ratios.

    Updates:
    --------
    ExplicitDamageRatio: dict[str, float]
        The explicit damage ratio for each component is updated based on the calculated damage ratio.
    """

    for component_type, component_resistance in self.ResistanceValues.items():
        # Calculate damage ratio for component, given resistance value matrix and component type.
        component_dr = np.count_nonzero(component_resistance == 0) / component_resistance.size

        # For RC and RS, there is a thresholds, 35%. if dr > 35%, then set dr as 1, totaled.
        # Florida loss model VIII page 32(49)
        if (component_dr > 0.35) and (component_type in ['Roof Coverings', 'Roof Sheathings']):
            component_dr = 1

        # 2012/09/21, change damage ratio for wall assumption.
        # 1, check 4 walls 3 status, for each wall, one status failed means this wall failed.
        # 2, if one wall fails, consider drW = 1.
        # if component is wall

        # if component_type == 'Walls':
        #     WallFail = 0
        #     component_resistance = np.reshape(component_resistance, (4, -1))
        #     for i in range(component_resistance.shape[0]):
        #         Wall = component_resistance[i, :]
        #         if np.any(Wall == 0):
        #             WallFail += 1
        #     # if two walls fail, consider wall totaled (4 walls fail).
        #     if WallFail >= 2:
        #         WallFail = 4
        #     component_dr = 0.25 * WallFail

        if component_type == 'Walls':
            WallFail = np.sum(np.any(component_resistance == 0))
            # if two walls fail, consider wall totaled (4 walls fail).
            if WallFail >= 2:
                WallFail = 4
            component_dr = 0.25 * WallFail

        self.ExplicitDamageRatioWind[component_type] = component_dr

    def fn_damage_ratio_flood(self):
    """
    This method calculates the flood damage ratio based on the flood depth and the damage state for flood.

    Attributes:
    -----------
    self:
        FloodDepthIndex: int
            The index of the flood depth.
        ProbDS_Flood: np.ndarray
            The probability of the damage state for flood.
        DamageState_Flood: np.ndarray
            The damage state for flood.
        ImplicitDamageRatioFlood: float
            The implicit damage ratio due to flood.

    Updates:
    --------
    ImplicitDamageRatioFlood: float
        The implicit damage ratio due to flood is updated based on the calculated damage ratio.
    """

    tempProb = self.tempProbDS_Flood[self.FloodDepthIndex, :]
    # Choose one DS.
    fflag = np.where(np.random.rand() <= tempProb)[0]
    if fflag.size == 0:
        fDS = 0
    else:
        fDS = fflag[0]
    # Random one damage value as % of value of bldg in this DS range.
    if fDS == 0:
        # if DS = 0, no damage
        bdrFlood = 0
    elif fDS == self.DamageState_Flood.shape[0] - 1:
        # if DS = 6, more than 40%, consider as totalled, set as 70%.
        bdrFlood = 0.7
    else:
        # Random generate one damage ratio between the range specified by damage states.
        bdrFloodRange = (self.DamageState_Flood[fDS, 1] + np.random.permutation(
            self.DamageState_Flood[fDS + 1, 1] - self.DamageState_Flood[fDS, 1]) - 1) / 100
        bdrFlood = bdrFloodRange[0]

    self.ImplicitDamageRatioFlood = bdrFlood

    def fn_damage_ratio_interior_wind(self):
    """
    This method calculates the interior damage ratio based on the damage ratios of various components of the
    building. Reference: FPHLM Volume III, p35(52).

    Attributes:
    -----------
    self:
        ExplicitDamageRatioWind: dict[str, float]
            A dictionary that holds the explicit damage ratio for each component. The keys are the component types and the values are the damage ratios.
        ImplicitDamageRatioWind: dict[str, float]
            A dictionary that holds the implicit damage ratio for each component due to wind. The keys are the component types and the values are the damage ratios.

    Updates:
    --------
    ImplicitDamageRatioWind: dict[str, float]
        The implicit damage ratio for each component due to wind is updated based on the calculated damage ratio.
    """

    # 2-story bldg adjustment ratio.
    k2 = 1.1

    # 1-story bldg
    if self.Building.Story == 1:
        # Eq. for RC is wrong in p35, but right in p46.
        tempRC = 0.62 * self.ExplicitDamageRatioWind["Roof Coverings"] ** 2 - 0.02 * self.ExplicitDamageRatioWind[
            "Roof Coverings"]
        tempRS = 1.29 * self.ExplicitDamageRatioWind["Roof Sheathings"]
        tempRWC = 0.38 * self.ExplicitDamageRatioWind["Roof-Wall Connections"] ** 2 + 0.23 * self.ExplicitDamageRatioWind[
            "Roof-Wall Connections"]
        tempW = 7.91 * self.ExplicitDamageRatioWind["Walls"] ** 3 - 8.7 * self.ExplicitDamageRatioWind[
            "Walls"] ** 2 + 4.21 * self.ExplicitDamageRatioWind["Walls"]
        tempO = 0.39 * self.ExplicitDamageRatioWind["Openings"] ** 2 + 0.31 * self.ExplicitDamageRatioWind["Openings"]
    else:
        tempRC = k2 * (0.62 * self.ExplicitDamageRatioWind["Roof Coverings"] ** 2 - 0.02 * self.ExplicitDamageRatioWind[
            "Roof Coverings"])
        tempRS = k2 * (1.29 * self.ExplicitDamageRatioWind["Roof Sheathings"])
        tempRWC = k2 * (
                0.38 * self.ExplicitDamageRatioWind["Roof-Wall Connections"] ** 2 + 0.23 * self.ExplicitDamageRatioWind[
            "Roof-Wall Connections"])
        tempW = k2 * (7.91 * self.ExplicitDamageRatioWind["Walls"] ** 3 - 8.7 * self.ExplicitDamageRatioWind[
            "Walls"] ** 2 + 4.21 * self.ExplicitDamageRatioWind["Walls"])
        tempO = k2 * (
                0.39 * self.ExplicitDamageRatioWind["Openings"] ** 2 + 0.31 * self.ExplicitDamageRatioWind["Openings"])

    # get mean of interior damage.
    drInteriorMean = max([tempRC, tempRS, tempRWC, tempW, tempO])

    # Interior damage follows Weibull distribution.
    # FPHLM Volume III, p37(54).
    alpha = 0.7854
    beta = 2
    drInterior = weibull_min.rvs(beta, scale=alpha) * drInteriorMean

    # cap drInterior within [0, 1].
    # FPHLM Volume III, p38(55).
    if drInterior >= 1.0:
        drInterior = 1.0
    elif drInterior <= 0.001:
        drInterior = 0

    self.ImplicitDamageRatioWind["Interior"] = drInterior

    def fn_damage_ratio_utility_wind(self):
    """
    This method calculates the utility damage ratio based on the damage ratios of various components of the building.

    Attributes:
    -----------
    self:
        ExplicitDamageRatioWind: dict[str, float]
            A dictionary that holds the explicit damage ratio for each component. The keys are the component types and the values are the damage ratios.
        ImplicitDamageRatioWind: dict[str, float]
            A dictionary that holds the implicit damage ratio for each component due to wind. The keys are the component types and the values are the damage ratios.

    Updates:
    --------
    ImplicitDamageRatioWind: dict[str, float]
        The implicit damage ratio for each component due to wind is updated based on the calculated damage ratio.
    """

    # Reference: FPHLM Volume III, p61(78).
    # utility damage should be max(k*each interior damage equation).

    # Decide coefficient k
    k_dict = {"Electrical": 0.5, "Mechanical": 0.4, "Plumbing": 0.35}

    for utility_type, k in k_dict.items():
        tempRC = k * (0.62 * self.ExplicitDamageRatioWind["Roof Coverings"] ** 2 - 0.2 * self.ExplicitDamageRatioWind[
            "Roof Coverings"])
        tempRS = k * (1.29 * self.ExplicitDamageRatioWind["Roof Sheathings"])
        tempRWC = k * (
                0.38 * self.ExplicitDamageRatioWind["Roof-Wall Connections"] ** 2 + 0.23 * self.ExplicitDamageRatioWind[
            "Roof-Wall Connections"])
        tempW = k * (7.91 * self.ExplicitDamageRatioWind["Walls"] ** 3 - 8.7 * self.ExplicitDamageRatioWind[
            "Walls"] ** 2 + 4.21 * self.ExplicitDamageRatioWind["Walls"])
        tempO = k * (0.39 * self.ExplicitDamageRatioWind["Openings"] ** 2 + 0.31 * self.ExplicitDamageRatioWind["Openings"])

        drUtilityMean = max([tempRC, tempRS, tempRWC, tempW, tempO])

        # FPHLM Volumn III, p37(54).
        alpha = 0.7854
        beta = 2
        drUtility = weibull_min.rvs(beta, scale=alpha) * drUtilityMean

        # FPHLM Volumn III, p62(79).
        if drUtility >= 1:
            drUtility = 1
        elif drUtility <= 0.001:
            drUtility = 0

        self.ImplicitDamageRatioWind[utility_type] = drUtility

    def get_damage_ratio_implicit(self) -> float:
    """
    This method calculates the implicit damage ratio for the building.

    Attributes:
    -----------
    self:
        ImplicitDamageRatioWind: dict[str, float]
            A dictionary that holds the implicit damage ratio for each component due to wind. The keys are the component types and the values are the damage ratios.
        ImplicitDamageRatioFlood: float
            The implicit damage ratio due to flood.
        ReplacementRatio: dict[str, float]
            A dictionary that holds the replacement ratio for each component. The keys are the component types and the values are the replacement ratios.

    Returns:
    --------
    float:
        The implicit damage ratio for the building.
    """

    # Implicit damage ratio is weighted result of damage ratio of wind-implicit and flood.
    bdrWind_Im = self.get_damage_ratio_wind_implicit()
    bdrFlood = self.get_damage_ratio_flood_implicit()
    return min((max(bdrWind_Im, bdrFlood) + 0.5 * min(bdrWind_Im, bdrFlood)), 1)
