    @staticmethod
def calculate_resistance_values(num_component_array, resistance_value_c_index):
    """
    Calculate resistance values based on the component number and resistance value index.

    Parameters:
    -----------
    num_component_array : list
        The number of each component.
    resistance_value_c_index : np.ndarray
        The resistance value index array.
    ResistanceType : str
        The type of resistance ('Mean' or 'CV').

    Returns:
    --------
    resistance_array : np.ndarray
        The calculated resistance array.
    """
    resistance_array = np.concatenate([
        np.ones(num_component_array[0]) * resistance_value_c_index[0],
        np.ones(num_component_array[1]) * resistance_value_c_index[1],
        np.ones(num_component_array[2]) * resistance_value_c_index[2],
        np.ones(num_component_array[3]) * resistance_value_c_index[4],
        np.ones(num_component_array[4]) * resistance_value_c_index[5],
        np.ones(num_component_array[5]) * resistance_value_c_index[6],
        np.ones(num_component_array[6]) * resistance_value_c_index[7],
        np.ones(num_component_array[7]) * resistance_value_c_index[8],
        np.ones(num_component_array[8]) * resistance_value_c_index[9]
    ])

    return resistance_array

    @staticmethod
def fn_damage_probability(iteration_result, damage_states):
    """
    Calculate probability of damage states based on provided iteration results.

    Args:
        iteration_result (np.ndarray): The iteration results array [windSpeed, floodDepth, iteration]
        damage_states (np.ndarray): The damage states thresholds

    Returns:
        np.ndarray: Probability of damage states
    """
    # Initialize prob_ds_owfd with zeros
    prob_ds_owfd = np.zeros((iteration_result.shape[0], iteration_result.shape[1], damage_states.shape[0]))

    # Calculate the sum of tempDS == 0 for all i, j
    prob_ds_owfd[:, :, 0] = np.sum(iteration_result == 0, axis=2)

    # Create a 3D array for the conditions
    conditions = (iteration_result[..., np.newaxis] > damage_states[:-1, 1]) & \
                 (iteration_result[..., np.newaxis] <= damage_states[1:, 1])

    # Calculate the sum of conditions for all i, j, k
    prob_ds_owfd[:, :, 1:] = np.sum(conditions, axis=2)

    # Normalize by the maximum number of iterations
    prob_ds_owfd /= Config.NUM_OF_ITERATIONS

    return prob_ds_owfd

    @staticmethod
def initialize_resistance_values(ComponentCombination_cIndex, resistance_array):
    """
    Initialize resistance values based on the component combination index.

    Parameters:
    -----------
    ComponentCombination_cIndex : np.ndarray
        The component combination index array.
    resistance_array : np.ndarray
        The resistance array to be initialized.

    Returns:
    --------
    resistance_array : np.ndarray
        The initialized resistance array.
    """
    # Initialize resistance_array with zeros
    resistance_array_out = np.zeros((resistance_array.shape[0],))
    # Roof cover (RC)
    resistance_array_out[0] = resistance_array[0, ComponentCombination_cIndex[0]-1]
    # Roof sheathing (RS)
    resistance_array_out[1] = resistance_array[1, ComponentCombination_cIndex[1]-1]
    # Roof-to-wall connection (RWC)
    resistance_array_out[2:4] = resistance_array[2:4, ComponentCombination_cIndex[2]-1]
    # Opening (O)
    resistance_array_out[4:7] = resistance_array[4:7, ComponentCombination_cIndex[3]-1]
    # Wall (W)
    resistance_array_out[7:10] = resistance_array[7:10, ComponentCombination_cIndex[4]-1]

    return resistance_array_out

    def aggregate_results(self, output_dir: str = None) -> Dict[Tuple[str, int], np.ndarray]:
    """
    Aggregates all individual building type and cIndex result files into a single dictionary.
    Populates the self.ProbDS_wfd_dict property.

    Args:
        output_dir (str, optional): Directory containing the individual result files.
                                   Defaults to Config.OUTPUTS_PATH.

    Returns:
        Dict[Tuple[str, int], np.ndarray]: Dictionary containing all damage probability arrays
                                          keyed by (building_type, cindex).
    """
    if output_dir is None:
        output_dir = Config.OUTPUTS_PATH

    logging.info(f"Aggregating result files from {output_dir}")

    # Prepare the aggregate dictionary
    self.ProbDS_wfd_dict = {}

    # List all files in the directory with the appropriate pattern
    file_pattern = f'ProbDS_wfd_m*_c*_iter{Config.NUM_OF_ITERATIONS}.pkl'
    file_list = [f for f in os.listdir(output_dir) 
                 if f.startswith('ProbDS_wfd_m') and f.endswith(f'_iter{Config.NUM_OF_ITERATIONS}.pkl')]

    logging.info(f"Found {len(file_list)} result files to aggregate")

    for filename in file_list:
        # Extract building type and cIndex from the filename
        # Format is ProbDS_wfd_m{building_type}_c{cindex}_iter{iterations}.pkl
        try:
            # Parse the filename
            parts = filename.split('_')
            m_part = parts[2].replace('m', '')
            c_part = parts[3].replace('c', '')

            building_type = int(m_part)
            cindex = int(c_part)

            # Only aggregate if we should process this combination (if filters are set)
            if not self.should_process_combination(building_type, cindex):
                logging.info(f"Skipping aggregation of {filename} based on user filters")
                continue

            # Load the file
            file_path = os.path.join(output_dir, filename)
            with open(file_path, 'rb') as file:
                probds_wfd = pickle.load(file)

            # Add to the aggregate dictionary
            self.ProbDS_wfd_dict[(building_type, cindex)] = probds_wfd
            logging.info(f"Loaded {filename} for building type {building_type} and cIndex {cindex}")

        except Exception as e:
            logging.error(f"Error processing file {filename}: {str(e)}")

    # Optionally save the aggregated dictionary
    aggregated_file_path = os.path.join(output_dir, f'ProbDS_wfd_dictMaxIter{Config.NUM_OF_ITERATIONS}_aggregated.pkl')
    with open(aggregated_file_path, 'wb') as file:
        pickle.dump(self.ProbDS_wfd_dict, file)

    logging.info(f"Aggregated {len(self.ProbDS_wfd_dict)} results into {aggregated_file_path}")

    return self.ProbDS_wfd_dict

    def get_available_building_types(self) -> List[int]:
    """

    building_types = []

    def process_component_combination(self, building: Building, ComponentCombination, cIndex: int) -> tuple:
    """
    Process the component combination for a building with parallelization over iterations.

    This method processes a specific component configuration for a building across all
    wind orientations, wind speeds, and flood depths using Monte Carlo simulation with
    parallel processing over iterations.

    Parameters:
        building (Building): The building object to process.
        ComponentCombination (np.ndarray): Array of valid component combinations for
            this building's roof shape.
        cIndex (int): The 1-based component configuration index to process.

    Returns:
        tuple: A tuple containing (building.BldgType, cIndex, averaged_damage_probability_array),
            or None if this combination should be skipped based on user filters.

    Info:
        When Config.EXPORT_DETAILED_DAMAGE_RATIOS is True, this method also exports
        detailed damage ratio breakdowns to a CSV file for each (building_type, cIndex)
        combination. The CSV file contains damage ratios averaged over Monte Carlo
        iterations for each (wind_orientation, wind_speed, flood_depth) combination.

        The CSV file is saved to Config.OUTPUTS_PATH with the naming pattern:
        DetailedDR_m{building_type}_c{cIndex}_iter{NUM_OF_ITERATIONS}.csv
    """
    logging.info(f"CIndex: {cIndex}")
    logging.info(f"Building type: {building.BldgType}")

    if cIndex < 1 or cIndex > self.ComponentCombination_All.shape[0]:
        raise ValueError(
            f"cIndex {cIndex} is out of range. Expected a 1-based index in "
            f"[1, {self.ComponentCombination_All.shape[0]}]."
        )

    # Convert public 1-based cIndex to internal 0-based array index.
    component_config_index = cIndex - 1

    # Check if this combination should be processed
    if not self.should_process_combination(building.BldgType, cIndex):
        logging.info(f"Skipping building {building.BldgType} with cIndex {cIndex} based on user filters")
        return None

    # Get the wind-related components combination for current configuration
    ComponentCombination_cIndex = self.ComponentCombination_All[component_config_index, 0:5]

    # Prepare an array for storing damage probabilities for each wind orientation
    num_orient = len(self.WindOrientations)
    local_ProbDS_owfd = np.zeros((num_orient, len(self.WindSpeeds), self.ProbDS_Flood.shape[2],
                                 len(self.DamageStates)), dtype=np.float32)

    # Determine if we should collect detailed damage ratios
    collect_detailed = Config.EXPORT_DETAILED_DAMAGE_RATIOS

    # Initialize the exporter if collecting detailed ratios
    exporter = None
    if collect_detailed:
        exporter = DamageRatioExporter(building_type=building.BldgType, cindex=cIndex)
        logging.info(f"Collecting detailed damage ratios for building {building.BldgType}, cIndex {cIndex}")

    # Only allow valid component combinations
    if np.any(np.all(ComponentCombination == ComponentCombination_cIndex, axis=1)):
        # Initialize resistance parameters locally
        MeanResistanceValue_cIndex = self.initialize_resistance_values(ComponentCombination_cIndex,
                                                                     self.MeanResistanceValue)
        MeanResistanceCV_cIndex = self.initialize_resistance_values(ComponentCombination_cIndex,
                                                                  self.MeanResistanceCV)

        # Determine the number of components
        NumRC = np.sum(building.NumPanel)
        NumRS = np.sum(building.NumPanel)
        NumRWC = np.sum(building.NumConnection)
        NumO_Door = building.DoorFront + building.DoorBack
        NumO_Garage = building.DoorGarage
        NumO_Window = building.OpeningFront + building.OpeningBack + 2 * (building.OpeningSideH + building.OpeningSideR)
        NumW_Shear = 4
        NumW_Uplift = 4
        NumW_OutOfPlane = 4
        NumComponent = [NumRC, NumRS, NumRWC, NumO_Door + NumO_Garage + NumO_Window,
                      NumW_Shear + NumW_Uplift + NumW_OutOfPlane]
        NumComponentArray = [NumRC, NumRS, NumRWC, NumO_Door, NumO_Garage, NumO_Window,
                           NumW_Shear, NumW_Uplift, NumW_OutOfPlane]

        tempMean = self.calculate_resistance_values(NumComponentArray, MeanResistanceValue_cIndex)
        tempCV = self.calculate_resistance_values(NumComponentArray, MeanResistanceCV_cIndex)

        # Generate LHS samples
        Sigma = self.Covariance.create_cov_matrix(NumComponent, tempMean, tempCV)
        ComponentSample = self.Sampling.generate_lhs_samples_norm(Config.NUM_OF_ITERATIONS, tempMean, Sigma)

        # Define the ranges for wind speeds and flood depths once
        wIndex_range = range(len(self.WindSpeeds))
        fIndex_range = range(self.ProbDS_Flood.shape[2])

        # Process each wind orientation independently
        for oIndex in range(num_orient):
            WindOrientation_oIndex = self.WindOrientations[oIndex, 1]
            # Local iteration results: [windSpeed, flood depth, iteration]
            local_IterationResult = np.zeros((len(self.WindSpeeds), self.ProbDS_Flood.shape[2], 
                                           Config.NUM_OF_ITERATIONS))

            # Parallelize over iterations
            parallel_results = Parallel(n_jobs=Config.NUM_OF_WORKERS, mmap_mode='r+', verbose=1)(
                delayed(self.process_single_iteration)(
                    ComponentSample, Iter, building, WindOrientation_oIndex, 
                    NumComponent, wIndex_range, fIndex_range, component_config_index, collect_detailed
                ) for Iter in range(Config.NUM_OF_ITERATIONS)
            )

            # Reshape results into the local_IterationResult array
            # and collect detailed results if requested
            for Iter, (iter_result, detailed_list) in enumerate(parallel_results):
                for w_idx in range(len(wIndex_range)):
                    for f_idx in range(len(fIndex_range)):
                        local_IterationResult[w_idx, f_idx, Iter] = iter_result[w_idx, f_idx]

                # Add detailed results to exporter if collecting
                if collect_detailed and detailed_list is not None:
                    for detailed_dict in detailed_list:
                        exporter.add_record(
                            wind_orientation_index=oIndex,
                            wind_speed_index=detailed_dict["wind_speed_index"],
                            flood_depth_index=detailed_dict["flood_depth_index"],
                            iteration=Iter,
                            detailed_ratios={k: v for k, v in detailed_dict.items() 
                                           if k not in ["wind_speed_index", "flood_depth_index"]}
                        )

            # Compute damage probability from local iteration results
            local_ProbDS_owfd[oIndex, :, :, :] = self.fn_damage_probability(local_IterationResult, self.DamageStates)

        # Average over orientations
        local_ProbDS_wfdo = np.transpose(local_ProbDS_owfd, (1, 2, 3, 0))
        ProbDS_wfd = np.mean(local_ProbDS_wfdo, axis=3)

        # Save the result immediately to disk
        self.save_result_per_building_and_cindex(building.BldgType, cIndex, ProbDS_wfd)

        # Save detailed damage ratios to CSV if collecting
        if collect_detailed and exporter is not None:
            exporter.aggregate_and_save_to_csv()
            exporter.clear()  # Free memory

        return building.BldgType, cIndex, ProbDS_wfd

    else:
        # If the combination is not valid for this roof shape, save a zero probability array.
        # This occurs for hip roof buildings (odd BldgType) where only 96 of 192 configurations
        # match. Gable roof buildings (even BldgType) match all 192 configurations.
        logging.debug(
            f"Component combination cIndex {cIndex} is not valid for building type "
            f"{building.BldgType} (RoofShape={building.RoofShape}). Saving zero array."
        )
        default_array = np.zeros((len(self.WindSpeeds), self.ProbDS_Flood.shape[2], len(self.DamageStates)), dtype=np.float32)

        # Save the empty result to disk as well
        self.save_result_per_building_and_cindex(building.BldgType, cIndex, default_array)

        return building.BldgType, cIndex, default_array

    def process_single_iteration(self, ComponentSample, Iter, building, WindOrientation_oIndex, NumComponent, wIndex_range, fIndex_range, cIndex, collect_detailed: bool = False):
    """
    Process a single iteration for a given orientation, building, and component combination.

    Parameters:
        ComponentSample (np.ndarray): Array of component resistance samples.
        Iter (int): The current iteration number.
        building (Building): The building object being processed.
        WindOrientation_oIndex (float): The wind orientation angle value.
        NumComponent (List[int]): List containing the number of each component type.
        wIndex_range (range): Range of wind speed indices to process.
        fIndex_range (range): Range of flood depth indices to process.
        cIndex (int): The 0-based component configuration index into ComponentCombination_All.
        collect_detailed (bool): If True, collect detailed component-level damage ratios
            for CSV export. Default is False. This is controlled by
            Config.EXPORT_DETAILED_DAMAGE_RATIOS.

    Raises:
        IndexError: If cIndex is out of bounds for ComponentCombination_All.

    Returns:
        tuple: A tuple containing:
            - numpy.ndarray: Iteration results for this iteration across all wind speeds
                and flood depths, shape (len(wIndex_range), len(fIndex_range)).
            - List[Dict[str, Any]] or None: If collect_detailed is True, a list of
                dictionaries containing detailed damage ratio breakdowns for each
                (wind_speed, flood_depth) combination. Each dict includes keys for
                wind_speed_index, flood_depth_index, and all component damage ratios.
                If collect_detailed is False, returns None.

    Info:
        When collect_detailed is True, the detailed ratios include:
            - dr_total: Total building damage ratio
            - dr_explicit: Explicit damage ratio
            - dr_implicit: Implicit damage ratio
            - dr_exp_wind_RC, dr_exp_wind_RS, dr_exp_wind_RWC, dr_exp_wind_O, dr_exp_wind_W:
                Explicit wind damage ratios for each component
            - dr_imp_wind_I, dr_imp_wind_E, dr_imp_wind_M, dr_imp_wind_P:
                Implicit wind damage ratios for each component
            - dr_imp_flood: Implicit flood damage ratio
    """
    # Validate cIndex is within bounds of ComponentCombination_All
    num_configs = self.ComponentCombination_All.shape[0]
    if not (0 <= cIndex < num_configs):
        raise IndexError(
            f"cIndex {cIndex} is out of bounds for ComponentCombination_All "
            f"with {num_configs} rows. Expected a 0-based index in [0, {num_configs - 1}]."
        )

    # Extract resistance values for current sample
    rRC = ComponentSample[Iter, :NumComponent[0]]
    rRS = ComponentSample[Iter, NumComponent[0]:sum(NumComponent[:2])]
    rRWC = ComponentSample[Iter, sum(NumComponent[:2]):sum(NumComponent[:3])]
    rO = ComponentSample[Iter, sum(NumComponent[:3]):sum(NumComponent[:4])]
    rW = ComponentSample[Iter, sum(NumComponent[:4]):]

    # Transpose arrays for calculations
    rRCt, rRSt, rRWCt, rOt, rWt = rRC.T, rRS.T, rRWC.T, rO.T, rW.T

    # Results array for this iteration
    iter_results = np.zeros((len(wIndex_range), len(fIndex_range)))

    # Detailed results list (only populated if collect_detailed is True)
    detailed_results = [] if collect_detailed else None

    # Create the FailureChecks object for this iteration
    failure_checks = LoadingAndFailureCheck(rRSt, rRCt, rRWCt, rOt, rWt,
                                         building, WindOrientation_oIndex,
                                         self.WindSpeeds[0, 1])

    # Loop over wind speeds
    for w_idx, wIndex in enumerate(wIndex_range):
        WindSpeed_wIndex = self.WindSpeeds[wIndex, 1]
        if wIndex > 0:
            failure_checks.update_inputs_and_outputs(rRSt, rRCt, rRWCt, rOt, rWt,
                                                  building, WindOrientation_oIndex,
                                                  WindSpeed_wIndex)

        # Initialize the DamageRatios object
        damage_ratios = DamageRatioCalculation(Building=building,
                                            FailureChecks=failure_checks,
                                            ComponentConfigurationIndex=cIndex,
                                            FloodDepthIndex=0)

        # Loop over flood depths
        for f_idx, fIndex in enumerate(fIndex_range):
            if fIndex > 0:
                damage_ratios.FloodDepthIndex = fIndex
                damage_ratios.fn_damage_ratio_flood()

            # Store the result
            iter_results[w_idx, f_idx] = damage_ratios.get_building_damage_ratio()

            # Collect detailed ratios if requested
            if collect_detailed:
                detailed_dict = DamageRatioExporter.collect_detailed_ratios(damage_ratios)
                detailed_dict["wind_speed_index"] = wIndex
                detailed_dict["flood_depth_index"] = fIndex
                detailed_results.append(detailed_dict)

    return iter_results, detailed_results

    def should_process_combination(self, building_type: int, cindex: int) -> bool:
    """
    Determine if this building type and cindex combination should be processed
    based on user-specified filters.

    Args:
        building_type (int): Building type identifier
        cindex (int): 1-based component configuration index

    Returns:
        bool: True if this combination should be processed, False otherwise
    """
    # If no filters are set, process all combinations

    building_type_match = (self.building_types_to_process is None or 
                           building_type in self.building_types_to_process)

    # Check cindex filter
    cindex_match = (self.cindex_to_process is None or 
                    cindex in self.cindex_to_process)

    # Both filters must pass
    return building_type_match and cindex_match

    def aggregate_and_save_to_csv(self, output_dir: Optional[str] = None) -> str:
    """
    Aggregate collected data by averaging over iterations and save to CSV.

    This method groups the collected damage ratio records by
    (wind_orientation_index, wind_speed_index, flood_depth_index), computes
    the mean of all damage ratio columns over iterations, and saves the
    result to a CSV file.

    Parameters:
        output_dir (str, optional): Directory path for output file.
            Defaults to Config.OUTPUTS_PATH if not provided.

    Returns:
        str: The file path of the saved CSV file.

    Info:
        The CSV file is named using the pattern:
        DetailedDR_m{building_type}_c{cindex}_iter{NUM_OF_ITERATIONS}.csv

        The 'iteration' column is excluded from the output as values are
        averaged over all iterations.
    """
    if output_dir is None:
        output_dir = Config.OUTPUTS_PATH

    # Create output directory if it doesn't exist
    os.makedirs(output_dir, exist_ok=True)

    if not self.collected_data:
        logging.warning(
            f"No data collected for building type {self.building_type}, cIndex {self.cindex}. "
            "Skipping CSV export."
        )
        return ""

    # Convert to DataFrame
    df = pd.DataFrame(self.collected_data)

    # Group by index columns and compute mean over iterations
    group_columns = ["m", "c", "wind_orientation_index", "wind_speed_index", "flood_depth_index"]

    # Get all damage ratio columns (everything except group columns and iteration)
    dr_columns = [col for col in df.columns if col not in group_columns and col != "iteration"]

    # Aggregate by taking mean over iterations
    aggregated_df = df.groupby(group_columns, as_index=False)[dr_columns].mean()

    # Define column order for output
    column_order = [
        "m", "c", "wind_orientation_index", "wind_speed_index", "flood_depth_index",
        "dr_total", "dr_explicit", "dr_implicit",
        "dr_exp_wind_RC", "dr_exp_wind_RS", "dr_exp_wind_RWC", "dr_exp_wind_O", "dr_exp_wind_W",
        "dr_imp_wind_I", "dr_imp_wind_E", "dr_imp_wind_M", "dr_imp_wind_P",
        "dr_imp_flood"
    ]

    # Reorder columns
    aggregated_df = aggregated_df[column_order]

    # Generate filename
    file_name = f"DetailedDR_m{self.building_type}_c{self.cindex}_iter{Config.NUM_OF_ITERATIONS}.csv"
    file_path = os.path.join(output_dir, file_name)

    # Save to CSV
    aggregated_df.to_csv(file_path, index=False)

    logging.info(
        f"Saved detailed damage ratios for building type {self.building_type}, "
        f"cIndex {self.cindex} to {file_path}"
    )

    return file_path

    @staticmethod
def collect_detailed_ratios(damage_ratios: DamageRatioCalculation) -> Dict[str, float]:
    """
    Extract detailed damage ratios from a DamageRatioCalculation object.

    This method extracts all component-level damage ratios from the given
    DamageRatioCalculation object and returns them in a dictionary with
    standardized column names for CSV export.

    Parameters:
        damage_ratios (DamageRatioCalculation): The damage ratio calculation object
            containing computed damage ratios for all components.

    Returns:
        Dict[str, float]: A dictionary containing all damage ratio values with keys:
            - dr_total: Total building damage ratio
            - dr_explicit: Explicit damage ratio
            - dr_implicit: Implicit damage ratio
            - dr_exp_wind_RC: Roof Coverings explicit damage ratio
            - dr_exp_wind_RS: Roof Sheathings explicit damage ratio
            - dr_exp_wind_RWC: Roof-Wall Connections explicit damage ratio
            - dr_exp_wind_O: Openings explicit damage ratio
            - dr_exp_wind_W: Walls explicit damage ratio
            - dr_imp_wind_I: Interior implicit damage ratio
            - dr_imp_wind_E: Electrical implicit damage ratio
            - dr_imp_wind_M: Mechanical implicit damage ratio
            - dr_imp_wind_P: Plumbing implicit damage ratio
            - dr_imp_flood: Flood implicit damage ratio

    Info:
        This method accesses the ExplicitDamageRatioWind, ImplicitDamageRatioWind,
        and ImplicitDamageRatioFlood attributes of the DamageRatioCalculation object
        to extract component-level values.
    """
    result = {
        "dr_total": damage_ratios.get_building_damage_ratio(),
        "dr_explicit": damage_ratios.get_damage_ratio_explicit(),
        "dr_implicit": damage_ratios.get_damage_ratio_implicit(),
        "dr_imp_flood": damage_ratios.get_damage_ratio_flood_implicit(),
    }

    # Add explicit wind component damage ratios
    for component_name, column_name in EXPLICIT_WIND_COLUMNS.items():
        result[column_name] = damage_ratios.ExplicitDamageRatioWind.get(component_name, 0.0)

    # Add implicit wind component damage ratios
    for component_name, column_name in IMPLICIT_WIND_COLUMNS.items():
        result[column_name] = damage_ratios.ImplicitDamageRatioWind.get(component_name, 0.0)

    return result
