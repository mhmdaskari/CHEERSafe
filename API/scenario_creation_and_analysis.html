    def calculate_period_probability(self):
    """
    Calculate probability of hurricane occurrence in each time period

    Based on the MATLAB function Scenario_PeriodCalculation
    """
    # Get hurricane probabilities
    hurricane_prob = self.hurricane_data['Hurricane_Prob']

    # Load hurricane count data
    hurricane_count = Utils.from_json('Inputs/HurricaneCount.json')

    # Calculate normalized daily probability
    hurricane_daily_prob = hurricane_count[:, 3] / np.sum(hurricane_count[:, 3])

    # Calculate cumulative daily probability
    hurricane_daily_cum_prob = np.column_stack((
        hurricane_count[:, 0],  # Week ID
        hurricane_count[:, 2],  # Day ID
        np.cumsum(hurricane_daily_prob)  # Cumulative probability
    ))

    # Calculate conditional probability of each hurricane in each year
    hurricane_conditional_prob = np.outer(hurricane_daily_prob, hurricane_prob[:, 1])

    # Calculate uniform probability for time periods
    uniform_prob = np.cumsum(np.ones(self.time_periods) / self.time_periods)

    # Calculate starting time of each time period
    period_starts = np.zeros((self.time_periods + 1, 3))
    period_starts[0, :] = [hurricane_count[0, 0], hurricane_count[0, 2], 0]
    period_starts[-1, :] = [hurricane_count[-1, 0], hurricane_count[-1, 2], len(hurricane_count)]

    # Find intermediate period starts
    for i in range(1, self.time_periods):
        for j in range(len(hurricane_daily_cum_prob) - 1):
            if (hurricane_daily_cum_prob[j, 2] <= uniform_prob[i-1] and
                hurricane_daily_cum_prob[j+1, 2] >= uniform_prob[i-1]):
                period_starts[i, 0] = hurricane_daily_cum_prob[j, 0]
                period_starts[i, 1] = period_starts[0, 1] + j
                period_starts[i, 2] = j
                break

    # Calculate probability of each hurricane in each time period
    hurricane_period_prob = np.zeros((self.time_periods, len(hurricane_prob)))
    for i in range(self.time_periods):
        for j in range(int(period_starts[i, 2]) + 1, int(period_starts[i+1, 2])):
            hurricane_period_prob[i, :] += hurricane_conditional_prob[j, :]

    hurricane_period_prob = hurricane_period_prob.T

    # Store results
    self.hurricane_period_prob = hurricane_period_prob
    self.period_starts = period_starts[:-1]  # Remove the last row

    return hurricane_period_prob, period_starts[:-1]

    def simulate_hurricanes(self, num_scenarios: int) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Simulate hurricane occurrences over the specified number of scenarios

    Parameters:
        num_scenarios (int): Number of scenarios to simulate

    Returns:
        Tuple[np.ndarray, np.ndarray, np.ndarray]: 
            - Scenario matrix (scenarios x time periods)
            - Hurricane IDs for each scenario 
            - Inter-arrival times between hurricanes
    """
    # Total time periods in one scenario
    total_periods = self.years * self.time_periods

    # Replicate hurricane period probability for all years
    hurricane_period_prob_scenario = np.tile(self.hurricane_period_prob, (1, self.years))

    # Initialize output arrays
    scenarios = np.zeros((num_scenarios, total_periods + 2))
    hurricane_ids = np.zeros((num_scenarios, 50))  # Assuming max 50 hurricanes per scenario
    inter_arrival_times = np.zeros_like(hurricane_ids)

    # For each scenario
    for s in range(num_scenarios):
        # For each time period
        for t in range(total_periods):
            # Calculate cumulative probability for all hurricanes at this period
            # 1-97: hurricane; 98: no-hurricane
            temp_prob = np.append(
                hurricane_period_prob_scenario[:, t], 
                1 - np.sum(hurricane_period_prob_scenario[:, t])
            )
            temp_prob = np.cumsum(temp_prob)

            # Randomly select hurricane based on probability
            random_value = np.random.rand()
            hurricane_idx = np.where(temp_prob >= random_value)[0][0]

            # Record hurricane index
            scenarios[s, t] = hurricane_idx + 1  # 1-indexed

        # Find which time periods have hurricanes
        hurricane_periods = np.where(scenarios[s, :-2] < 98)[0]

        # Count number of hurricanes in scenario
        scenarios[s, -1] = len(hurricane_periods)

        # Record hurricane IDs
        if len(hurricane_periods) > 0:
            hurricane_ids[s, :len(hurricane_periods)] = scenarios[s, hurricane_periods]

            # Calculate inter-arrival times
            temp_ia = np.zeros(len(hurricane_periods))
            temp_ia[0] = hurricane_periods[0] + 1  # First arrival time

            for j in range(1, len(hurricane_periods)):
                temp_ia[j] = hurricane_periods[j] - hurricane_periods[j-1]

            inter_arrival_times[s, 1:len(temp_ia)+1] = temp_ia
            inter_arrival_times[s, 0] = np.mean(temp_ia) if len(temp_ia) > 0 else 0

    return scenarios, hurricane_ids, inter_arrival_times

    def generate_report(self, save_dir: Optional[str] = None) -> Dict:
    """
    Generate analysis report with plots

    Parameters:
        save_dir (str, optional): Directory to save plots, defaults to output_dir

    Returns:
        Dict: Dictionary containing paths to saved plots
    """
    if save_dir is None:
        save_dir = self.output_dir

    # Run analysis
    results = self.run_analysis()

    # Create plots
    plot_paths = {}

    # Total loss CDF plot
    total_loss_path = f'{save_dir}/Fig1_TotalLoss_{self.num_scenarios}Scenarios.jpg'
    self.plot_total_loss_cdf(save_path=total_loss_path)
    plot_paths['total_loss'] = total_loss_path

    # Hurricane count plot
    hurricane_count_path = f'{save_dir}/Fig2_NumHurricane_{self.num_scenarios}Scenarios.jpg'
    self.plot_hurricane_count(save_path=hurricane_count_path)
    plot_paths['hurricane_count'] = hurricane_count_path

    # Inter-arrival time plot
    inter_arrival_time_path = f'{save_dir}/Fig3_InterArrivalTime_{self.num_scenarios}Scenario.jpg'
    self.plot_inter_arrival_time(save_path=inter_arrival_time_path)
    plot_paths['inter_arrival_time'] = inter_arrival_time_path

    # Annual probability plot
    annual_probability_path = f'{save_dir}/Fig4_AnnualProb_{self.num_scenarios}Scenario.jpg'
    self.plot_annual_probability(save_path=annual_probability_path)
    plot_paths['annual_probability'] = annual_probability_path

    # Save results
    results_path = self.save_results(results)

    return {
        'results_file': results_path,
        'plots': plot_paths
    }

    def run_analysis(self) -> Dict:
    """
    Run the complete scenario analysis workflow

    Returns:
        Dict: Dictionary containing analysis results
    """
    # Get basic analysis results
    analysis_results = self.analyze_scenarios()

    # Calculate losses for scenarios
    scenario_losses = self.calculate_losses(self.scenarios)

    # Add to analysis results
    analysis_results['losses'] = {
        'total': scenario_losses.sum(),
        'mean': scenario_losses.mean(),
        'std': scenario_losses.std(),
        'min': scenario_losses.min(),
        'max': scenario_losses.max(),
        'per_scenario': scenario_losses.tolist()
    }

    return analysis_results

    def save_results(self, results: Dict = None, file_name: Optional[str] = None) -> str:
    """
    Save analysis results to file

    Parameters:
        results (Dict, optional): Analysis results to save, runs analysis if None
        file_name (str, optional): File name to save results, defaults to auto-generated name

    Returns:
        str: Path where results were saved
    """
    if results is None:
        results = self.run_analysis()

    if file_name is None:
        file_name = f'ScenarioAnalysis_{self.num_scenarios}.json'

    file_path = f'{self.output_dir}/{file_name}'

    # Convert numpy arrays to lists for JSON serialization
    for key in results:
        if isinstance(results[key], np.ndarray):
            results[key] = results[key].tolist()
        elif isinstance(results[key], dict):
            for subkey in results[key]:
                if isinstance(results[key][subkey], np.ndarray):
                    results[key][subkey] = results[key][subkey].tolist()

    # Save to file using Utils.to_json
    from ..tools import Utils
    Utils.to_json(results, file_path)

    return file_path

    def analyze_scenarios(self) -> Dict:
    """
    Analyze simulated hurricane scenarios

    Returns:
        Dict: Dictionary containing analysis results
    """
    # Calculate expected loss (mean) for hurricane set
    hurricane_prob = self.hurricane_data['Hurricane_Prob']
    expected_loss_annual = np.sum(self.hurricane_loss[:, 1] * hurricane_prob[:, 1])
    expected_loss_scenario = expected_loss_annual * self.years

    # Calculate standard deviation for hurricane set
    std_loss_annual = np.sqrt(np.sum((self.hurricane_loss[:, 1] - expected_loss_annual)**2 * hurricane_prob[:, 1]))
    std_loss_scenario = std_loss_annual * np.sqrt(self.years)

    # Calculate statistics for number of hurricanes
    # Assumed as Poisson distribution
    num_hurricane_lambda = np.sum(hurricane_prob[:97, 1]) * self.years

    # Calculate statistics for inter-arrival time
    # Assumed as Exponential distribution
    inter_arrival_time_lambda = 1 / np.sum(hurricane_prob[:97, 1]) * self.time_periods

    # Return analysis results
    return {
        'expected_loss': {
            'annual': expected_loss_annual,
            'scenario': expected_loss_scenario
        },
        'std_loss': {
            'annual': std_loss_annual,
            'scenario': std_loss_scenario
        },
        'hurricane_count': {
            'lambda': num_hurricane_lambda
        },
        'inter_arrival_time': {
            'lambda': inter_arrival_time_lambda
        },
        'scenario_data': {
            'scenarios': self.scenarios,
            'hurricane_ids': self.hurricane_ids,
            'inter_arrival_times': self.inter_arrival_times
        }
    }

    def plot_annual_probability(self, save_path: Optional[str] = None) -> plt.Figure:
    """
    Plot scatter of annual probability

    Parameters:
        save_path (str, optional): Path to save the plot

    Returns:
        plt.Figure: Figure object
    """
    plt.figure(figsize=(10, 6))

    # Get hurricane probabilities
    hurricane_prob = self.hurricane_data['Hurricane_Prob'][:97, 1]

    # Calculate empirical hurricane probabilities
    temp_hurricane_id = self.hurricane_ids.flatten()
    temp_hurricane_id = temp_hurricane_id[temp_hurricane_id > 0]

    # Count occurrences of each hurricane
    hurricane_counts = np.zeros(97)
    for i in range(1, 98):  # 1-indexed
        hurricane_counts[i-1] = np.sum(temp_hurricane_id == i)

    # Calculate empirical probabilities
    total_years = self.years * self.num_scenarios
    hurricane_empirical_prob = hurricane_counts / total_years

    # Plot scatter
    plt.scatter(hurricane_prob, hurricane_empirical_prob)

    # Plot diagonal line
    x = np.linspace(0, 0.04, 100)
    plt.plot(x, x)

    plt.xlim(0, 0.04)
    plt.ylim(0, 0.04)
    plt.xticks(np.arange(0, 0.05, 0.01))
    plt.yticks(np.arange(0, 0.05, 0.01))
    plt.xlabel('All possible scenarios', fontsize=14)
    plt.ylabel('Simulated scenarios', fontsize=14)

    if save_path:
        plt.savefig(save_path)

    return plt.gcf()

    def calculate_losses(self, scenarios: np.ndarray) -> np.ndarray:
    """
    Calculate losses for simulated hurricane scenarios

    Parameters:
        scenarios (np.ndarray): Simulated hurricane scenarios

    Returns:
        np.ndarray: Hurricane losses for each scenario
    """
    num_scenarios = scenarios.shape[0]
    scenario_losses = np.zeros(num_scenarios)

    for s in range(num_scenarios):
        # Get hurricane IDs for this scenario (excluding no-hurricane periods)
        hurricane_ids = scenarios[s, :-2][scenarios[s, :-2] < 98].astype(int)

        # Sum losses for all hurricanes in this scenario
        for h_id in hurricane_ids:
            # Use h_id - 1 as index since hurricane IDs are 1-indexed
            scenario_losses[s] += self.loss_estimator.hurricane_losses['HURR'][h_id - 1]

    return scenario_losses

    def setup_loss_estimator(self):
    """
    Set up the loss estimator for regional losses
    """
    # Create the regional loss estimator using the loss_estimation module
    self.loss_estimator = RegionalLossEstimator(
        hurricane_data_path=self.hurricane_data_path,
        inventory_path=self.inventory_path,
        damage_state_path='Inputs/DamageState.json',
        wind_speed_path='Inputs/WindSpeed.json',
        flood_depth_path='Inputs/FloodDepth.json',
        peril='HURR'
    )

    # Make sure regional losses are calculated
    if self.loss_wind_path and os.path.exists(self.loss_wind_path):
        self.loss_estimator.hurricane_losses['WIND'] = Utils.from_json(self.loss_wind_path)
    else:
        # Use full loss estimation module to calculate and save wind losses
        loss_estimation = LossEstimation(
            hurricane_data_path=self.hurricane_data_path,
            inventory_path=self.inventory_path,
            prob_ds_file_path=self.prob_ds_file_path
        )
        wind_result = loss_estimation.estimate_regional_losses(peril='WIND')
        self.loss_estimator.hurricane_losses['WIND'] = wind_result['regional_losses']

    if self.loss_flood_path and os.path.exists(self.loss_flood_path):
        self.loss_estimator.hurricane_losses['FLOOD'] = Utils.from_json(self.loss_flood_path)
    else:
        # Use full loss estimation module to calculate and save flood losses
        loss_estimation = LossEstimation(
            hurricane_data_path=self.hurricane_data_path,
            inventory_path=self.inventory_path,
            prob_ds_file_path=self.prob_ds_file_path
        )
        flood_result = loss_estimation.estimate_regional_losses(peril='FLOOD')
        self.loss_estimator.hurricane_losses['FLOOD'] = flood_result['regional_losses']

    # Combine for hurricane losses
    hurr_result = self.loss_estimator.combine_wind_flood_losses()
    self.loss_estimator.hurricane_losses['HURR'] = hurr_result
